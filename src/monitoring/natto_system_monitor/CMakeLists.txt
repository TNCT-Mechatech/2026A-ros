cmake_minimum_required(VERSION 3.16)
project(natto_system_monitor)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp diagnostic_msgs diagnostic_updater diagnostic_aggregator pluginlib sensor_msgs geometry_msgs nav_msgs std_msgs natto_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(system_monitor src/system_monitor.cpp)
target_include_directories(system_monitor PRIVATE include)
ament_target_dependencies(system_monitor rclcpp diagnostic_updater diagnostic_msgs sensor_msgs geometry_msgs nav_msgs std_msgs natto_msgs)
add_library(health_analyzers SHARED src/health_analyzers.cpp)
target_include_directories(health_analyzers PRIVATE include)
ament_target_dependencies(health_analyzers diagnostic_aggregator pluginlib geometry_msgs sensor_msgs)
pluginlib_export_plugin_description_file(diagnostic_aggregator analyzers.xml)
install(TARGETS system_monitor DESTINATION lib/${PROJECT_NAME})
install(TARGETS health_analyzers LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md INVESTIGATION.md DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_dependencies(rclcpp diagnostic_updater diagnostic_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_monitor_integration test/test_monitor_integration.py TIMEOUT 90)
  ament_add_gtest(test_serial_observer test/test_serial_observer.cpp)
  target_include_directories(test_serial_observer PRIVATE include)
  ament_target_dependencies(test_serial_observer rclcpp diagnostic_msgs diagnostic_updater)
  set_tests_properties(test_serial_observer PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=197;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
  ament_add_gtest(test_health test/test_health.cpp)
  target_include_directories(test_health PRIVATE include)
  ament_target_dependencies(test_health diagnostic_msgs geometry_msgs sensor_msgs)
endif()
ament_package()
